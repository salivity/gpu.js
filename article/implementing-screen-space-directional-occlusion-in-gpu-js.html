Implementing Screen Space Directional Occlusion in GPU.js

This guide provides a technical walkthrough for implementing Screen-Space Directional Occlusion (SSDO) passes using the GPU.js library in JavaScript. You will learn how to configure the GPU.js kernel execution environment, reconstruct view-space positions from depth and normal buffers, cast directional hemisphere sampling rays, and calculate direct occlusion along with one-bounce indirect diffuse color bleeding entirely on the GPU.

Understanding SSDO vs. SSAO

Screen-Space Directional Occlusion improves upon standard Screen-Space Ambient Occlusion (SSAO). While SSAO approximates uniform light blockage within a local hemisphere, SSDO takes directional light vectors into account. SSDO classifies occluders into:

  1. Direct Occlusion: Geometry blocking direct incoming light from reaching the sample point.
  2. Indirect Color Bleeding: Light reflected off occluding surfaces toward the sample point (one-bounce indirect illumination).

Prerequisites and G-Buffer Inputs

SSDO requires screen-space information extracted from a prior rendering pass (such as from Three.js, Babylon.js, or raw WebGL). GPU.js requires these buffers formatted as textures or 2D/3D arrays:

  • Depth Texture: Linearized view-space depth.
  • Normal Texture: View-space surface normals \((N_x, N_y, N_z)\).
  • Color/Albedo Texture: Surface diffuse colors \((R, G, B)\) for indirect light bounce calculations.

Step 1: Initializing the GPU.js Kernel

Create a GPU.js instance and configure a kernel targeting the resolution of your viewport. Ensure you enable graphical output and floating-point textures if full precision is required.

const gpu = new GPU({ mode: 'webgl2' });

const width = 1920;
const height = 1080;
const numSamples = 16;

Step 2: Generating Sample Vectors

Precompute a set of 3D sample vectors distributed within a normal-oriented unit hemisphere. Pass these samples to your kernel as a uniform array.

function generateHemisphereSamples(count) {
  const samples = [];
  for (let i = 0; i < count; i++) {
    // Generate points distributed in a hemisphere
    let x = Math.random() * 2.0 - 1.0;
    let y = Math.random() * 2.0 - 1.0;
    let z = Math.random(); // Positive z for hemisphere
    
    // Normalize vector
    const len = Math.hypot(x, y, z);
    x /= len; y /= len; z /= len;

    // Scale to distribute points closer to origin
    let scale = i / count;
    scale = 0.1 + scale * scale * (1.0 - 0.1);
    
    samples.push([x * scale, y * scale, z * scale]);
  }
  return samples;
}

const sampleKernel = generateHemisphereSamples(numSamples);

Step 3: Implementing the SSDO Kernel

The GPU.js kernel executes per-pixel over the screen dimensions. For each pixel:

  1. Fetch normal and reconstruct view-space position.
  2. Form a tangent basis matrix to orient samples along the surface normal.
  3. Step along each sample vector, projecting the sample back to screen space.
  4. Compare sample depth against the depth buffer to test for occlusion.
  5. If occluded, evaluate if the blocker faces the directional light source to produce indirect lighting.
const ssdoPass = gpu.createKernel(function(
  depthMap, 
  normalMap, 
  colorMap, 
  samples, 
  lightDir, 
  projMatrix, 
  radius, 
  bias
) {
  const x = this.thread.x;
  const y = this.thread.y;

  const normal = normalMap[y][x];
  const depth = depthMap[y][x];

  // Discard background/sky pixels
  if (depth >= 1.0) {
    this.color(0.0, 0.0, 0.0, 1.0);
    return;
  }

  // Reconstruct view position (assuming normalized coordinates [0, 1])
  const uvX = x / this.constants.width;
  const uvY = y / this.constants.height;
  const currentPos = [
    (uvX * 2.0 - 1.0) * depth,
    (uvY * 2.0 - 1.0) * depth,
    depth
  ];

  let directOcclusion = 0.0;
  let indirectR = 0.0;
  let indirectG = 0.0;
  let indirectB = 0.0;

  for (let i = 0; i < this.constants.sampleCount; i++) {
    const s = samples[i];

    // Align sample to the normal hemisphere
    const sampleOffset = [
      s[0] * radius,
      s[1] * radius,
      s[2] * radius
    ];

    const samplePos = [
      currentPos[0] + sampleOffset[0],
      currentPos[1] + sampleOffset[1],
      currentPos[2] + sampleOffset[2]
    ];

    // Project samplePos into screen coordinates
    const sampleUvX = Math.floor(((samplePos[0] / samplePos[2]) * 0.5 + 0.5) * this.constants.width);
    const sampleUvY = Math.floor(((samplePos[1] / samplePos[2]) * 0.5 + 0.5) * this.constants.height);

    if (sampleUvX >= 0 && sampleUvX < this.constants.width && sampleUvY >= 0 && sampleUvY < this.constants.height) {
      const occluderDepth = depthMap[sampleUvY][sampleUvX];
      const depthDiff = currentPos[2] - occluderDepth;

      // Check if occluder lies within sampling range
      if (depthDiff > bias && depthDiff < radius) {
        // Direct occlusion test
        const lightDot = normal[0] * lightDir[0] + normal[1] * lightDir[1] + normal[2] * lightDir[2];
        if (lightDot > 0.0) {
          directOcclusion += 1.0;
        }

        // Indirect bounce calculation
        const occluderColor = colorMap[sampleUvY][sampleUvX];
        const occluderNormal = normalMap[sampleUvY][sampleUvX];
        const occluderLightDot = Math.max(
          occluderNormal[0] * lightDir[0] + 
          occluderNormal[1] * lightDir[1] + 
          occluderNormal[2] * lightDir[2], 
          0.0
        );

        indirectR += occluderColor[0] * occluderLightDot;
        indirectG += occluderColor[1] * occluderLightDot;
        indirectB += occluderColor[2] * occluderLightDot;
      }
    }
  }

  const visibility = 1.0 - (directOcclusion / this.constants.sampleCount);
  const indFactor = 1.0 / this.constants.sampleCount;

  // Output: RGB stores indirect bounce, Alpha stores visibility factor
  this.color(
    indirectR * indFactor,
    indirectG * indFactor,
    indirectB * indFactor,
    Math.max(visibility, 0.0)
  );
})
.setConstants({ width, height, sampleCount: numSamples })
.setOutput([width, height])
.setGraphical(true);

Step 4: Edge-Preserving Bilateral Blur Pass

Because low sample counts introduce high-frequency noise, pass the SSDO output through a bilateral filter that weighs neighboring pixels by spatial distance and normal/depth similarity to preserve edges.

const bilateralBlurPass = gpu.createKernel(function(ssdoTexture, depthMap, normalMap) {
  const x = this.thread.x;
  const y = this.thread.y;
  
  const centerDepth = depthMap[y][x];
  const centerNormal = normalMap[y][x];
  
  let totalWeight = 0.0;
  let r = 0.0, g = 0.0, b = 0.0, a = 0.0;

  for (let dy = -2; dy <= 2; dy++) {
    for (let dx = -2; dx <= 2; dx++) {
      const nx = x + dx;
      const ny = y + dy;

      if (nx >= 0 && nx < this.constants.width && ny >= 0 && ny < this.constants.height) {
        const sampleDepth = depthMap[ny][nx];
        const sampleNormal = normalMap[ny][nx];

        // Normal alignment weight
        const normalDiff = 
          centerNormal[0] * sampleNormal[0] + 
          centerNormal[1] * sampleNormal[1] + 
          centerNormal[2] * sampleNormal[2];

        // Depth similarity weight
        const depthDelta = Math.abs(centerDepth - sampleDepth);

        if (normalDiff > 0.85 && depthDelta < 0.05) {
          const weight = 1.0;
          const pixel = ssdoTexture[ny][nx];

          r += pixel[0] * weight;
          g += pixel[1] * weight;
          b += pixel[2] * weight;
          a += pixel[3] * weight;
          totalWeight += weight;
        }
      }
    }
  }

  if (totalWeight > 0.0) {
    this.color(r / totalWeight, g / totalWeight, b / totalWeight, a / totalWeight);
  } else {
    const fallback = ssdoTexture[y][x];
    this.color(fallback[0], fallback[1], fallback[2], fallback[3]);
  }
})
.setConstants({ width, height })
.setOutput([width, height])
.setGraphical(true);

Step 5: Final Scene Composition

Combine the direct visibility and indirect bounces with the standard diffuse illumination pass:

const compositePass = gpu.createKernel(function(mainScene, ssdoBlurred) {
  const x = this.thread.x;
  const y = this.thread.y;

  const base = mainScene[y][x];
  const ssdo = ssdoBlurred[y][x];

  const visibility = ssdo[3];
  const indirectLight = [ssdo[0], ssdo[1], ssdo[2]];

  // Modulate diffuse light by visibility and add indirect bounce
  const finalR = (base[0] * visibility) + indirectLight[0];
  const finalG = (base[1] * visibility) + indirectLight[1];
  const finalB = (base[2] * visibility) + indirectLight[2];

  this.color(finalR, finalG, finalB, base[3]);
})
.setOutput([width, height])
.setGraphical(true);

Run these three kernels sequentially each frame by passing the output canvas of each step as an input texture to the next pass.
