Bloom Effect in GPU.js: Downsampling and Blur

This article provides a step-by-step guide to implementing a high-performance bloom post-processing effect in GPU.js using sequential downsampling and separable blur passes. You will learn how to extract high-luminance areas, progressively scale down the resolution to efficiently simulate wide light dispersal, apply horizontal and vertical Gaussian blurs, and composite the blurred bloom texture back onto the original frame.

1. The Bloom Pipeline Architecture

A standard bloom pipeline consists of four major stages:

  1. Brightness Thresholding: Isolating the brightest pixels in the scene while discarding dark and mid-tone regions.
  2. Sequential Downsampling: Progressively halving the resolution of the thresholded texture into a pyramid of smaller mip levels.
  3. Separable Blur: Blurring each downsampled level using a two-pass (horizontal and vertical) Gaussian filter to maintain high performance.
  4. Additive Composition: Upsampling, combining the blurred levels, and adding the resulting bloom map back over the original input texture.

2. Setting Up GPU.js and the Brightness Pass

The first kernel isolates areas of the source image that exceed a specified brightness threshold using perceptual luminance coefficients: \(Y = 0.2126R + 0.7152G + 0.0722B\).

const { GPU } = require('gpu.js');
const gpu = new GPU();

const extractBrightPass = gpu.createKernel(function(image, width, height, threshold) {
    const x = this.thread.x;
    const y = this.thread.y;
    
    // Sample RGBA values normalized to [0, 1]
    const r = image[y][x][0];
    const g = image[y][x][1];
    const b = image[y][x][2];
    
    const luminance = 0.2126 * r + 0.7152 * g + 0.0722 * b;
    
    if (luminance > threshold) {
        this.color(r, g, b, 1.0);
    } else {
        this.color(0.0, 0.0, 0.0, 1.0);
    }
})
.setOutput([width, height])
.setGraphical(true)
.setPipeline(true);

3. Sequential Downsampling

Applying a large blur to a full-resolution texture requires an immense number of texture samples. Sequential downsampling reduces texture dimensions by half at each step, allowing a small blur kernel to cover a larger physical footprint of the image.

function createDownsampleKernel(inWidth, inHeight) {
    const outWidth = Math.floor(inWidth / 2);
    const outHeight = Math.floor(inHeight / 2);

    return gpu.createKernel(function(inputTexture) {
        const x = this.thread.x * 2;
        const y = this.thread.y * 2;

        // 4-tap box filter downsample
        const p00 = inputTexture[y][x];
        const p10 = inputTexture[y][x + 1];
        const p01 = inputTexture[y + 1][x];
        const p11 = inputTexture[y + 1][x + 1];

        const r = (p00[0] + p10[0] + p01[0] + p11[0]) * 0.25;
        const g = (p00[1] + p10[1] + p01[1] + p11[1]) * 0.25;
        const b = (p00[2] + p10[2] + p01[2] + p11[2]) * 0.25;

        this.color(r, g, b, 1.0);
    })
    .setOutput([outWidth, outHeight])
    .setGraphical(true)
    .setPipeline(true);
}

By chaining this kernel 3 to 5 times, you produce an array of progressively smaller textures (e.g., \(1/2\), \(1/4\), \(1/8\), \(1/16\)).

4. Separable Gaussian Blur

A 2D Gaussian blur of radius \(N\) requires \(N^2\) texture fetches per pixel. By separating the blur into a horizontal pass followed by a vertical pass, the complexity drops to \(2N\) fetches.

function createBlurKernel(width, height, isHorizontal) {
    return gpu.createKernel(function(texture) {
        const x = this.thread.x;
        const y = this.thread.y;

        // 5-tap Gaussian weights: [0.06136, 0.24477, 0.38774, 0.24477, 0.06136]
        const w0 = 0.38774;
        const w1 = 0.24477;
        const w2 = 0.06136;

        let r = 0.0, g = 0.0, b = 0.0;

        if (this.constants.horizontal === 1) {
            const c  = texture[y][x];
            const l1 = texture[y][Math.max(0, x - 1)];
            const r1 = texture[y][Math.min(this.constants.width - 1, x + 1)];
            const l2 = texture[y][Math.max(0, x - 2)];
            const r2 = texture[y][Math.min(this.constants.width - 1, x + 2)];

            r = c[0] * w0 + (l1[0] + r1[0]) * w1 + (l2[0] + r2[0]) * w2;
            g = c[1] * w0 + (l1[1] + r1[1]) * w1 + (l2[1] + r2[1]) * w2;
            b = c[2] * w0 + (l1[2] + r1[2]) * w1 + (l2[2] + r2[2]) * w2;
        } else {
            const c  = texture[y][x];
            const d1 = texture[Math.max(0, y - 1)][x];
            const u1 = texture[Math.min(this.constants.height - 1, y + 1)][x];
            const d2 = texture[Math.max(0, y - 2)][x];
            const u2 = texture[Math.min(this.constants.height - 1, y + 2)][x];

            r = c[0] * w0 + (d1[0] + u1[0]) * w1 + (d2[0] + u2[0]) * w2;
            g = c[1] * w0 + (d1[1] + u1[1]) * w1 + (d2[1] + u2[1]) * w2;
            b = c[2] * w0 + (d1[2] + u2[2]) * w1 + (d2[2] + u2[2]) * w2;
        }

        this.color(r, g, b, 1.0);
    })
    .setConstants({ width, height, horizontal: isHorizontal ? 1 : 0 })
    .setOutput([width, height])
    .setGraphical(true)
    .setPipeline(true);
}

Apply this separable blur to each downsampled tier created in the previous step.

5. Progressive Upsampling and Blending

To prevent blocky artifacts, reconstruct the bloom effect by upsampling the smallest blurred layer into the next larger layer, summing their values until returning to full resolution.

function createUpsampleBlendKernel(destWidth, destHeight, srcWidth, srcHeight) {
    return gpu.createKernel(function(baseTexture, smallTexture) {
        const x = this.thread.x;
        const y = this.thread.y;

        const basePixel = baseTexture[y][x];

        // Bilinear coordinate mapping to the smaller texture
        const sx = (x / this.constants.destWidth) * this.constants.srcWidth;
        const sy = (y / this.constants.destHeight) * this.constants.srcHeight;
        
        const smallPixel = smallTexture[Math.floor(sy)][Math.floor(sx)];

        // Blend colors
        const r = basePixel[0] + smallPixel[0];
        const g = basePixel[1] + smallPixel[1];
        const b = basePixel[2] + smallPixel[2];

        this.color(r, g, b, 1.0);
    })
    .setConstants({ destWidth, destHeight, srcWidth, srcHeight })
    .setOutput([destWidth, destHeight])
    .setGraphical(true)
    .setPipeline(true);
}

6. Final Additive Composite Pass

Once the bloom map is back at full resolution, composite it additively with the original unprocessed frame, scaling by a bloom intensity factor.

const compositeKernel = gpu.createKernel(function(originalImage, bloomTexture, intensity) {
    const x = this.thread.x;
    const y = this.thread.y;

    const base = originalImage[y][x];
    const bloom = bloomTexture[y][x];

    const r = base[0] + bloom[0] * intensity;
    const g = base[1] + bloom[1] * intensity;
    const b = base[2] + bloom[2] * intensity;

    // Optional Tone-mapping (Reinhard) to prevent overexposure clipping
    const mappedR = r / (1.0 + r);
    const mappedG = g / (1.0 + g);
    const mappedB = b / (1.0 + b);

    this.color(mappedR, mappedG, mappedB, 1.0);
})
.setOutput([width, height]);

Execution Order Summary

  1. Execute extractBrightPass to get bright areas.
  2. Run downsampling passes sequentially to create a mip chain.
  3. Apply horizontal and vertical createBlurKernel passes to each mip level.
  4. Chain createUpsampleBlendKernel from the smallest mip back up to full size.
  5. Pass the base scene and reconstructed bloom texture into compositeKernel for output.
