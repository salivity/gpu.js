Distance Transform on Binary Masks Using GPU.js

The Euclidean Distance Transform (EDT) assigns each pixel in a binary segmentation mask a value corresponding to its distance to the nearest background pixel. While traditional CPU implementations like Felzenszwalb's algorithm run sequentially in linear time, running them in the browser or Node.js for high-resolution masks can bottleneck real-time applications. Utilizing GPU.js allows you to run parallelized WebGL kernels directly in JavaScript. This article explains how to adapt the Jump Flooding Algorithm (JFA)—a massively parallel algorithm ideal for GPUs—to compute the distance transform on binary masks using GPU.js.

Why Use the Jump Flooding Algorithm (JFA)?

Standard 1D-separable distance transform algorithms are difficult to map efficiently to SIMD/GPU architectures because they rely on sequential updates across rows and columns. The Jump Flooding Algorithm solves this by propagating nearest-coordinate data in parallel across all pixels simultaneously.

For an \(N \times N\) mask, JFA converges in \(\log_2(N)\) passes. In each pass \(k\), every pixel queries eight neighbors at an offset of \(step = N / 2^k\) (e.g., \(N/2, N/4, \dots, 1\)) and updates its stored nearest-seed coordinate if a neighbor points to a closer seed.

Pipeline Architecture in GPU.js

A high-performance implementation in GPU.js requires three distinct stages, chained together using pipeline: true so intermediate data stays in WebGL texture memory without costly CPU readbacks:

  1. Initialization Kernel: Converts the binary mask into an initial coordinate grid. Target pixels (foreground) are set to an invalid coordinate [-1, -1], while boundary/background pixels store their own coordinates [x, y].
  2. JFA Step Kernel: Repeatedly runs in a loop with decreasing step sizes. Each pixel evaluates its neighbors at the current step distance and keeps the coordinate with the smallest Euclidean distance.
  3. Distance Calculation Kernel: Takes the final coordinate map and calculates the actual Euclidean distance: \(\sqrt{(x - x_{seed})^2 + (y - y_{seed})^2}\).

Implementation

Below is the complete implementation of the distance transform using GPU.js.

const { GPU } = require('gpu.js'); // or import { GPU } from 'gpu.js' in the browser

function createDistanceTransform(width, height) {
  const gpu = new GPU();

  // 1. Initialize coordinate map: background stores [x, y], foreground stores [-1, -1]
  const initKernel = gpu.createKernel(function(mask) {
    const x = this.thread.x;
    const y = this.thread.y;
    
    // Mask value 0 is background (seed), 1 is foreground object
    if (mask[y][x] === 0) {
      return [x, y];
    } else {
      return [-1, -1];
    }
  })
  .setOutput([width, height])
  .setPipeline(true);

  // 2. JFA Step: Compares 8 neighbors at current step distance
  const jfaKernel = gpu.createKernel(function(coordTexture, step) {
    const x = this.thread.x;
    const y = this.thread.y;

    let bestX = coordTexture[y][x][0];
    let bestY = coordTexture[y][x][1];
    let bestDist = 9999999.0;

    if (bestX >= 0.0 && bestY >= 0.0) {
      const dx = x - bestX;
      const dy = y - bestY;
      bestDist = dx * dx + dy * dy;
    }

    // Check 8 neighbors
    for (let dy = -1; dy <= 1; dy++) {
      for (let dx = -1; dx <= 1; dx++) {
        if (dx === 0 && dy === 0) continue;

        const nx = x + dx * step;
        const ny = y + dy * step;

        if (nx >= 0 && nx < this.constants.width && ny >= 0 && ny < this.constants.height) {
          const seedX = coordTexture[ny][nx][0];
          const seedY = coordTexture[ny][nx][1];

          if (seedX >= 0.0 && seedY >= 0.0) {
            const diffX = x - seedX;
            const diffY = y - seedY;
            const dist = diffX * diffX + diffY * diffY;

            if (dist < bestDist) {
              bestDist = dist;
              bestX = seedX;
              bestY = seedY;
            }
          }
        }
      }
    }

    return [bestX, bestY];
  })
  .setConstants({ width, height })
  .setOutput([width, height])
  .setPipeline(true);

  // 3. Final Distance Kernel: Computes Euclidean distance from coordinates
  const finalDistanceKernel = gpu.createKernel(function(coordTexture) {
    const x = this.thread.x;
    const y = this.thread.y;

    const seedX = coordTexture[y][x][0];
    const seedY = coordTexture[y][x][1];

    if (seedX < 0.0 || seedY < 0.0) {
      return 0.0;
    }

    const dx = x - seedX;
    const dy = y - seedY;
    return Math.sqrt(dx * dx + dy * dy);
  })
  .setOutput([width, height]);

  // Execution wrapper
  return function compute(binaryMask) {
    let texture = initKernel(binaryMask);

    // Step sizes: powers of 2 starting at max(width, height) / 2 down to 1
    const maxDim = Math.max(width, height);
    let step = Math.pow(2, Math.floor(Math.log2(maxDim)));

    while (step >= 1) {
      const nextTexture = jfaKernel(texture, step);
      texture.delete(); // Free intermediate WebGL texture memory
      texture = nextTexture;
      step = Math.floor(step / 2);
    }

    const result = finalDistanceKernel(texture);
    texture.delete();
    return result;
  };
}

Usage Example

const width = 512;
const height = 512;

// Initialize the runner
const runDistanceTransform = createDistanceTransform(width, height);

// Example 512x512 mask where center circle is foreground (1) and outside is background (0)
const mask = Array.from({ length: height }, (_, y) =>
  Array.from({ length: width }, (_, x) => {
    const dx = x - width / 2;
    const dy = y - height / 2;
    return dx * dx + dy * dy < 10000 ? 1 : 0;
  })
);

// Compute distances
const distanceMap = runDistanceTransform(mask);
console.log(distanceMap);

Key Performance Guidelines

  • Pipeline Management: Always specify .setPipeline(true) on intermediate kernels. Without it, GPU.js reads pixel data back to the CPU after every jump-flooding step, degrading performance.
  • Memory Cleanups: Inside the execution loop, call .delete() on intermediate WebGL texture instances to prevent GPU memory leaks across iterations.
  • Aspect Ratios and Padding: JFA works most efficiently when the input dimensions are square powers of two. For arbitrary mask sizes, consider padding the input mask to the nearest power-of-two dimension and cropping the final distance map.
